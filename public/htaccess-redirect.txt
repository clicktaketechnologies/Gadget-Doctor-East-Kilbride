# ===== COPY THIS FILE TO cPanel public_html/.htaccess =====
# This permanently redirects gadgetdoctorls.co.uk → www.gadgetdoctorls.co.uk

RewriteEngine On

# Redirect non-www to www (301 permanent)
RewriteCond %{HTTP_HOST} ^gadgetdoctorls\.co\.uk$ [NC]
RewriteRule ^(.*)$ https://www.gadgetdoctorls.co.uk/$1 [R=301,L]

# Force HTTPS
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://www.gadgetdoctorls.co.uk/$1 [R=301,L]
