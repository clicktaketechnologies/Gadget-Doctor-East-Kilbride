    62→/** Today's date as yyyy-mm-dd, used as the min for the preferred date input. */
    63→function todayISO(): string {
    64→  const d = new Date();
    65→  const y = d.getFullYear();
    66→  const m = String(d.getMonth() + 1).padStart(2, "0");
    67→  const day = String(d.getDate()).padStart(2, "0");
    68→  return `${y}-${m}-${day}`;
    69→}
    70→
    71→function StepDots({
    72→  step,
    73→  total = TOTAL_STEPS,
    74→}: {
    75→  step: number;
    76→  total?: number;
    77→}) {
    78→  return (
    79→    <div className="flex items-center gap-1.5">
    80→      {Array.from({ length: total }).map((_, i) => {
    81→        const n = i + 1;
    82→        const done = n < step;
    83→        const active = n === step;
    84→        return (
    85→          <div
    86→            key={n}
    87→            className={cn(
    88→              "h-1.5 rounded-full transition-all",
    89→              done
    90→                ? "w-6 bg-primary"
    91→                : active
    92→                ? "w-8 bg-primary"
    93→                : "w-6 bg-muted"
    94→            )}
    95→          />
    96→        );
    97→      })}
    98→    </div>
    99→  );
   100→}

   130→

   138→        ? prev.filter((i) => i !== issue)

   144→    if (step === 1) return !!category;
   145→    if (step === 2) return !!deviceModel && (issues.length > 0 || description.trim().length > 0);

   148→      return collection === "dropoff" || (collection === "collection" && address.trim().length > 4);
   149→    const contactStep = collectionEnabled ? 4 : 3;

   151→      return (
   152→        name.trim().length > 1 &&
   153→        /.+@.+\..+/.test(email) &&
   154→        phone.replace(/\s+/g, "").length >= 7

   163→      description.trim() ? `Details: ${description.trim()}` : null,
   164→    ]
   165→      .filter(Boolean)
   166→      .join("\n");

   168→      customerName: name.trim(),
   169→      email: email.trim(),
   170→      phone: phone.trim(),
   171→      deviceType: category,
   172→      deviceModel,
   173→      issue: issueText || "No specific issue described",
   174→      needsCollection: collectionEnabled && collection === "collection",
   175→      collectionAddr:
   176→        collectionEnabled && collection === "collection"
   177→          ? address.trim()
   178→          : undefined,
   179→      bookingDate: bookingDate || undefined,
   180→      bookingTime: bookingTime || undefined,
   181→    });
   182→  };
   183→
   184→  if (submitted && ticketId) {
   185→    return (
   186→      <SuccessView
   187→        ticketId={ticketId}
   188→        name={name}
   189→        collection={collection}
   190→        onClose={onClose}
   191→      />
   192→    );
   193→  }
   194→
   195→  const next = () => setStep((s) => Math.min(totalSteps, s + 1));
   196→  const back = () => setStep((s) => Math.max(1, s - 1));
   197→
   198→  return (
   199→    <div className="flex flex-col">
   200→      {/* Stepper header */}
   201→      <div className="flex items-center justify-between gap-3 border-b border-border px-5 py-4 sm:px-6">

   203→          <DialogTitle className="text-base font-bold">
   204→            Book a Repair
   205→          </DialogTitle>
   206→          <DialogDescription className="text-xs">
   207→            Step {step} of {totalSteps} ·{" "}
   208→            {step === 1
   209→              ? "Choose device type"
   210→              : step === 2
   211→              ? "Describe the problem"
   212→              : step === 3 && collectionEnabled
   213→              ? "Collection preference"
   214→              : "Your contact details"}
   215→          </DialogDescription>
   216→        </div>
   217→        <StepDots step={step} total={totalSteps} />
   218→      </div>
   219→
   220→      {/* Body */}
   221→      <div className="custom-scroll max-h-[60vh] overflow-y-auto px-5 py-5 sm:px-6">

   338→        {/* Step 3: Collection preference (skipped when collection disabled) */}

   353→                  collection === "dropoff"

   358→                <RadioGroupItem

   362→                />
   363→                <div className="flex-1">

   750→export function BookingModal() {
   751→  const bookingOpen = useAppStore((s) => s.bookingOpen);
   752→  const bookingPreset = useAppStore((s) => s.bookingPreset);
   753→  const closeBooking = useAppStore((s) => s.closeBooking);
   754→  const { data: services } = useServices(true);
   755→
   756→  // Look up the service by preset slug to derive the category
   757→  const initialCategory = useMemo(() => {
   758→    if (!bookingPreset?.serviceSlug || !services) return null;
   759→    const svc = services.find((s) => s.slug === bookingPreset.serviceSlug);
   760→    if (svc) return svc.category;
   761→    // Allow direct deviceType override
   762→    if (bookingPreset.deviceType) return bookingPreset.deviceType;
   763→    return null;
   764→  }, [bookingPreset, services]);
   765→
   766→  return (
   767→    <Dialog
   768→      open={bookingOpen}
   769→      onOpenChange={(o) => {
   770→        if (!o) closeBooking();
   771→      }}
   772→    >
   773→      <DialogContent className="glass-strong max-h-[90vh] gap-0 overflow-hidden border-border p-0 sm:max-w-lg">
   774→        {/* key-remount: fresh local state every time preset changes / modal reopens */}
   775→        <BookingForm
   776→          key={bookingPreset?.serviceSlug ?? bookingPreset?.deviceType ?? "none"}
   777→          initialCategory={initialCategory}
   778→          onClose={closeBooking}
   779→        />
   780→      </DialogContent>
   781→    </Dialog>
   782→  );
   783→}
   784→
   785→export default BookingModal;
