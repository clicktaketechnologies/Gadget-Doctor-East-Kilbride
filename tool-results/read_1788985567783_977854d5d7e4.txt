   595→              hours.map((h, i) => (
   596→                <div
   597→                  key={i}
   598→                  className="grid grid-cols-1 gap-2 sm:grid-cols-[1fr_1.5fr_auto] sm:items-center"
   599→                >
   600→                  <Input
   601→                    aria-label={`Day ${i + 1}`}
   602→                    value={h.day}
   603→                    onChange={(e) => updateHour(i, "day", e.target.value)}
   604→                    placeholder="Monday – Thursday"
   605→                  />
   606→                  <Input
   607→                    aria-label={`Hours ${i + 1}`}
   608→                    value={h.time}
   609→                    onChange={(e) => updateHour(i, "time", e.target.value)}
   610→                    placeholder="09:30 AM – 03:00 PM"
   611→                  />
   612→                  <Button
   613→                    type="button"
   614→                    variant="ghost"
   615→                    size="icon"
   616→                    className="size-9 justify-self-start text-rose-300 hover:bg-rose-500/10 hover:text-rose-200 sm:justify-self-end"
   617→                    onClick={() => removeHour(i)}
   618→                    aria-label={`Remove hour row ${i + 1}`}
   619→                    disabled={updateMutation.isPending}
   620→                  >
   621→                    <Trash2 className="size-4" />
   622→                  </Button>
   623→                </div>
   624→              ))
   625→            )}
   626→            <Button
   627→              type="button"
   628→              variant="outline"
   629→              size="sm"
   630→              onClick={addHour}
   631→              className="mt-1 gap-1.5"
   632→            >
   633→              <Plus className="size-3.5" />
   634→              Add hour row
   635→            </Button>
   636→          </div>
   637→        </SectionCard>
   638→
   639→        {/* 6. Social Media */}
   640→        <SectionCard
   641→          icon={Share2}
   642→          title="Social Media"
   643→          subtitle="Links to your profiles"
   644→          full
   645→        >
   646→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   647→            {SOCIAL_LINKS.map((s) => (
   648→              <Field
   649→                key={s.key}
   650→                label={s.label}
   651→                htmlFor={`br-social-${s.key}`}
   652→              >
   653→                <div className="relative">
   654→                  <span className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-primary">
   655→                    <Icon name={s.icon} className="size-4" />
   656→                  </span>
   657→                  <Input
   658→                    id={`br-social-${s.key}`}
   659→                    value={socials[s.key] ?? ""}
   660→                    onChange={(e) => updateSocial(s.key, e.target.value)}
   661→                    placeholder={`https://${s.key}.com/your-page`}
   662→                    className="pl-9"
   663→                  />
   664→                </div>
   665→              </Field>
   666→            ))}
   667→          </div>
   668→        </SectionCard>
   669→
   670→        {/* 7. Target Areas */}
   671→        <SectionCard
   672→          icon={MapPinned}
   673→          title="Target Areas"
   674→          subtitle="Locations you serve"
   675→          full
   676→        >
   677→          <div className="space-y-2.5">
   678→            {targetAreas.length === 0 ? (
   679→              <div className="flex h-24 items-center justify-center rounded-lg border border-dashed border-border/60 text-xs text-muted-foreground">
   680→                No service areas yet. Add one below.
   681→              </div>
   682→            ) : (
   683→              targetAreas.map((area, i) => (
   684→                <div
   685→                  key={i}
   686→                  className="grid grid-cols-1 gap-2 sm:grid-cols-[1fr_auto] sm:items-center"
   687→                >
   688→                  <Input
   689→                    aria-label={`Service area ${i + 1}`}
   690→                    value={area}
   691→                    onChange={(e) => updateArea(i, e.target.value)}
   692→                    placeholder="e.g. East Kilbride"
   693→                  />
   694→                  <Button
   695→                    type="button"
   696→                    variant="ghost"
   697→                    size="icon"
   698→                    className="size-9 justify-self-start text-rose-300 hover:bg-rose-500/10 hover:text-rose-200 sm:justify-self-end"
   699→                    onClick={() => removeArea(i)}
   700→                    aria-label={`Remove area ${i + 1}`}
   701→                    disabled={updateMutation.isPending}
   702→                  >
   703→                    <Trash2 className="size-4" />
   704→                  </Button>
   705→                </div>
   706→              ))
   707→            )}
   708→            <Button
   709→              type="button"
   710→              variant="outline"
   711→              size="sm"
   712→              onClick={addArea}
   713→              className="mt-1 gap-1.5"
   714→            >
   715→              <Plus className="size-3.5" />
   716→              Add area
   717→            </Button>
   718→          </div>
   719→        </SectionCard>
   720→
   721→        {/* 8. Stats */}
   722→        <SectionCard
   723→          icon={Star}
   724→          title="Stats"
   725→          subtitle="Trust signals shown across the site"
   726→          full
   727→        >
   728→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   729→            <Field label="Google rating" htmlFor="br-rating" hint="0.0 – 5.0">
   730→              <Input
   731→                id="br-rating"
   732→                type="number"
   733→                min="0"
   734→                max="5"
   735→                step="0.1"
   736→                value={rating}
   737→                onChange={(e) => setRating(e.target.value)}
   738→                placeholder="4.6"
   739→              />
   740→            </Field>
   741→            <Field label="Review count" htmlFor="br-reviews">
   742→              <Input
   743→                id="br-reviews"
   744→                type="number"
   745→                min="0"
   746→                step="1"
   747→                value={reviewCount}
   748→                onChange={(e) => setReviewCount(e.target.value)}
   749→                placeholder="92"
   750→              />
   751→            </Field>
   752→            <Field label="Years of experience" htmlFor="br-years">
   753→              <Input
   754→                id="br-years"
   755→                type="number"
   756→                min="0"
   757→                step="1"
   758→                value={yearsExperience}
   759→                onChange={(e) => setYearsExperience(e.target.value)}
   760→                placeholder="12"
   761→              />
   762→            </Field>
   763→          </div>
   764→        </SectionCard>
   765→      </div>
   766→
   767→      {/* Sticky save bar */}
   768→      <div className="fixed inset-x-0 bottom-0 z-30 border-t border-border/60 bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80 lg:left-[260px]">
   769→        <div className="mx-auto flex max-w-6xl items-center justify-between gap-3 px-4 py-3 md:px-8">
   770→          <div className="flex min-w-0 items-center gap-2 text-sm">
   771→            {isDirty ? (
   772→              <>
   773→                <span className="size-2 shrink-0 rounded-full bg-amber-400" />
   774→                <span className="truncate text-muted-foreground">
   775→                  You have unsaved changes
   776→                </span>
   777→              </>
   778→            ) : savedFlash ? (
   779→              <>
   780→                <span className="size-2 shrink-0 rounded-full bg-emerald-400" />
   781→                <span className="truncate text-emerald-300">
   782→                  All changes saved
   783→                </span>
   784→              </>
   785→            ) : (
   786→              <>
   787→                <span className="size-2 shrink-0 rounded-full bg-emerald-400" />
   788→                <span className="truncate text-muted-foreground">
   789→                  All changes saved
   790→                </span>
   791→              </>
   792→            )}
   793→          </div>
   794→          <div className="flex shrink-0 items-center gap-2">
   795→            <a
   796→              href={website || "#"}
   797→              target="_blank"
   798→              rel="noopener noreferrer"
   799→              className="hidden items-center gap-1.5 text-xs text-muted-foreground transition-colors hover:text-foreground sm:inline-flex"
   800→            >
   801→              View live site
   802→              <ExternalLink className="size-3" />
   803→            </a>
   804→            <Button
   805→              variant="outline"
   806→              size="sm"
   807→              onClick={handleReset}
   808→              disabled={!isDirty || updateMutation.isPending}
   809→              className="gap-1.5"
   810→            >
   811→              <RotateCcw className="size-3.5" />
   812→              Reset
   813→            </Button>
   814→            <Button
   815→              size="sm"
   816→              onClick={handleSave}
   817→              disabled={!isDirty || updateMutation.isPending}
   818→              className="gap-1.5"
   819→            >
   820→              {updateMutation.isPending ? (
   821→                <>
   822→                  <Loader2 className="size-4 animate-spin" />
   823→                  Saving…
   824→                </>
   825→              ) : savedFlash ? (
   826→                <>
   827→                  <Check className="size-4" />
   828→                  Saved
   829→                </>
   830→              ) : (
   831→                <>
   832→                  <Save className="size-4" />
   833→                  Save Changes
   834→                </>
   835→              )}
   836→            </Button>
   837→          </div>
   838→        </div>
   839→      </div>
   840→    </div>
   841→  );
   842→}
